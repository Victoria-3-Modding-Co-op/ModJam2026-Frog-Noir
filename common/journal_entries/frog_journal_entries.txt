je_frog_noir_main = {
	group = je_group_global_international_situations
	icon = "gfx/interface/icons/event_icons/event_hardboiled.dds"
	weight = 100
	
	should_be_pinned_by_default_involved = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	widget = {
		gui = "gui/journal_entry_widgets/frog_journal_widgets.gui"
		name = "widget_frog_pi_firm"
		container = "custom_widget_container_je_icon"
	}
	
	widget = {
		gui = "gui/com_journal_injects/situation_widgets.gui"
		name = "widget_com_international_situation_involved_countries"
		container = "custom_widget_container_5"
	}
	
	scripted_button = frog_switch_focus_to_finding_clues_button
	scripted_button = frog_switch_focus_to_investigating_leads_button
	
	should_be_involved = {
		always = yes
	}
	
	immediate = {
		scope:journal_entry = {
			com_set_situation_illustration_background_effect = {
				texture = com_situation_background_europenorthamerica_london_center
			}
			com_situation_start_date_today_effect = yes
			set_variable = com_hide_involved_countries
			save_scope_as = je_frog_noir_main
		}
		set_global_variable = {
			name = je_frog_noir_main
			value = scope:je_frog_noir_main
		}
	}
	
	immediate_all_involved = {
		frog_initialize_local_firm_variables_effect = yes
	}
	
	on_become_involved_after_activation = {
		frog_initialize_local_firm_variables_effect = yes
	}
	
	complete = {
	}
	
	on_complete = {
		remove_global_variable = je_frog_noir_main
	}
	
	on_weekly_pulse = {
		effect = {
			scope:journal_entry = {
				every_scope_je_involved = {
					trigger_event = {
						on_action = frog_local_firm_weekly_pulse
					}
				}
			}
		}
	}
	
	on_monthly_pulse = {
		effect = {
			every_country = {
				limit = {
					NOT = {
						is_involved_in_journal_entry = je_frog_noir_main
					}
				}
				global_var:je_frog_noir_main = {
					add_involved_country = prev
				}
			}
		}
	}
}

je_frog_noir_investigating_leads = {
	group = je_group_internal_affairs
	icon = "gfx/interface/icons/event_icons/event_hardboiled.dds"
	weight = 100
	
	should_be_pinned_by_default_uninvolved_or_context = yes
	can_revolution_inherit = yes
	transferable = no
	can_deactivate = no
	
	scripted_progress_bar = frog_investigating_leads_progress_bar
	scripted_progress_bar = frog_investigating_leads_peril_bar
	
	immediate = {
		set_variable = {
			name = frog_investigation_progress_var
			value = 0
		}
		set_variable = {
			name = frog_investigation_peril_var
			value = 0
		}
		set_variable = {
			name = frog_investigation_timer_var
			value = 0
		}
	}
	
	complete = {
		custom_tooltip = {
			text = je_frog_noir_investigating_leads_complete
			var:frog_investigation_progress_var >= 8
		}
	}
	
	on_complete = {
		remove_variable = frog_investigation_progress_var
		remove_variable = frog_investigation_peril_var
	}
	
	fail = {
		custom_tooltip = {
			text = je_frog_noir_investigating_leads_fail
			var:frog_investigation_peril_var >= 5
		}
	}
	
	on_fail = {
		remove_variable = frog_investigation_progress_var
		remove_variable = frog_investigation_peril_var
	}
	
	invalid = {
		OR = {
			has_variable = frog_abandoned_investigation_var
			frog_country_has_official_investigator_trigger = no
		}
	}
	
	on_invalid = {
		remove_variable = frog_investigation_progress_var
		remove_variable = frog_investigation_peril_var
	}
	
	on_weekly_pulse = {
		events = {
		}
		random_events = {
		}
		effect = {
			frog_change_variable_effect = {
				NAME = frog_investigation_timer_var
				AMOUNT = 1
			}
			if = {
				limit = {
					var:frog_investigation_timer_var >= 20
				}
				frog_change_variable_effect = {
					NAME = frog_investigation_progress_var
					AMOUNT = 1
				}
				set_variable = {
					name = frog_investigation_timer_var
					value = 0
				}
			}
		}
	}
	
	status_desc = {
		desc = je_frog_noir_investigating_leads_status
	}
}
