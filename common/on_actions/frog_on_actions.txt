####---------- Game Started after Lobby ----------####

on_game_started_after_lobby = {
	on_actions = {
		frog_on_game_started_after_lobby
	}
}

frog_on_game_started_after_lobby = {
	effect = {
		every_country = {
			limit = {
				is_player = yes
			}
			trigger_event = {
				id = frog_intro_events.1
				popup = yes
			}
		}
	}
}

####---------- Country Yearly Pulses ----------####

on_yearly_pulse_country = {
	on_actions = {
		frog_on_yearly_pulse_country
	}
}

frog_on_yearly_pulse_country = {
	effect = {
		every_scope_character = {
			limit = {
				has_role = character_role_frog_private_investigator
			}
			frog_character_investigator_experience_gain_effect = { AMOUNT = frog_yearly_experience_gain_value }
		}
	}
}

####---------- Country Monthly Pulses ----------####

on_monthly_pulse_country = {
	on_actions = {
		frog_local_firm_monthly_pulse
		frog_monthly_events_pulse
	}
}

frog_local_firm_monthly_pulse = {
	effect = {
		clear_variable_list = frog_investigators_list
		every_scope_character = {
			limit = {
				has_role = character_role_frog_private_investigator
			}
			add_to_variable_list = {
				name = frog_investigators_list
				target = this
			}
		}
	}
}

frog_monthly_events_pulse = {
	random_events = {
		chance_of_no_event = 75
		5 = frog_frogged_by_toads_events.1
		5 = frog_frogged_by_toads_events.2
		5 = frog_frogged_by_toads_events.3
	}
}


####---------- Weekly Pulses for Journals ----------####
frog_local_firm_weekly_pulse = {
	on_actions = {
		frog_finding_clues_weekly_pulse
	}
	effect = {
		if = {
			limit = {
				has_variable = frog_accumulated_clues_var
				var:frog_accumulated_clues_var >= 10
			}
			frog_change_variable_effect = {
				NAME = frog_accumulated_clues_var
				AMOUNT = -10
			}
			frog_change_variable_effect = {
				NAME = frog_accumulated_leads_var
				AMOUNT = 1
			}
			post_notification = frog_gained_a_lead
		}
	}
}

frog_finding_clues_weekly_pulse = {
	trigger = {
		any_scope_character = {
			has_modifier = modifier_frog_sleuthing_for_clues
		}
	}
	events = {  }
	random_events = {
		chance_of_no_event = 50
		5 = frog_investigation_events.1
		5 = frog_investigation_events.2
	}
	effect = {
		frog_change_variable_effect = {
			NAME = frog_clue_generation_progress_var
			AMOUNT = frog_clue_generation_progress_value
		}
		if = {
			limit = {
				var:frog_clue_generation_progress_var >= 100
			}
			frog_add_clues_effect = {
				value = frog_clue_generation_value
			}
		}
	}
}
