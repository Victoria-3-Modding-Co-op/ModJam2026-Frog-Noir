frog_switch_focus_to_finding_clues_button = {
	name = frog_switch_focus_to_finding_clues_button
	desc = frog_placeholder_text

	visible = {
		is_involved_in_journal_entry = je_frog_noir_main
	}
	selected = {
		frog_country_is_focused_on_finding_clues_trigger = yes
	}
	possible = {
		frog_country_is_focused_on_finding_clues_trigger = no
	}
	effect = {
		var:frog_investigator_var = {
			add_modifier = modifier_frog_sleuthing_for_clues
			hidden_effect = {
				remove_modifier = modifier_frog_following_leads
				remove_modifier = modifier_frog_keeping_watch
			}
		}
		set_variable = {
			name = frog_abandoned_investigation_var
			months = 3
		}
	}
}

frog_switch_focus_to_investigating_leads_button = {
	name = frog_switch_focus_to_investigating_leads_button
	desc = frog_placeholder_text

	visible = {
		is_involved_in_journal_entry = je_frog_noir_main
	}
	selected = {
		frog_country_is_focused_on_following_leads_trigger = yes
	}
	possible = {
		NOT = {
			has_journal_entry = je_frog_noir_investigating_leads
		}
		custom_tooltip = {
			text = frog_has_one_or_more_leads
			var:frog_accumulated_leads_var >= 1
		}
	}
	effect = {
		var:frog_investigator_var = {
			add_modifier = modifier_frog_following_leads
			hidden_effect = {
				remove_modifier = modifier_frog_sleuthing_for_clues
				remove_modifier = modifier_frog_keeping_watch
			}
		}
		add_journal_entry = {
			type = je_frog_noir_investigating_leads
		}
	}
}

frog_recruit_new_investigator_button = {
	name = frog_recruit_new_investigator_button
	desc = frog_recruit_new_investigator_button_desc

	visible = {
		is_involved_in_journal_entry = je_frog_noir_main
	}

	possible = {
		custom_tooltip = {
			text = frog_has_less_than_3_pis_tt

			any_character_is_private_investigator = {
				count < 3
			}
		}
	}
	effect = {
		add_treasury = -1000000
		custom_tooltip = {
			text = frog_create_new_investigator_for_pool_tt
			
			frog_create_random_investigator_for_character_creation_effect = yes
			frog_remove_new_investigator_effect = yes
		}
		frog_update_investigator_list_effect = yes
	}
}
