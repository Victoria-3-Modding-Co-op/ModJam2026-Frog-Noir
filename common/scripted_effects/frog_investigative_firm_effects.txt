####---------- Frog Investigation Effects ----------####

# These effects are used to manage progress/peril in field investigations

frog_initialize_local_firm_effect = {
	set_variable = {
		name = frog_clue_generation_progress_var
		value = 0
	}
	set_variable = {
		name = frog_accumulated_clues_var
		value = 0
	}
	set_variable = {
		name = frog_accumulated_leads_var
		value = 0
	}
	trigger_event = frog_narrative_events.1
}

frog_initialize_investigation_variables_effect = {
	set_variable = {
		name = frog_investigation_progress_var
		value = 0
	}
	set_variable = {
		name = frog_investigation_peril_var
		value = 0
	}
	set_variable = {
		name = frog_investigation_timer_var
		value = 0
	}
}

frog_clear_investigation_variables_effect = {
	remove_variable = frog_investigation_progress_var
	remove_variable = frog_investigation_peril_var
	remove_variable = frog_investigation_timer_var
}

frog_complete_investigation_effect = {
	if = {
		limit = {
			global_var:frog_story_var ?= 1
		}
		trigger_event = frog_narrative_events.2
	}
	else_if = {
		limit = {
			global_var:frog_story_var ?= 2
		}
		trigger_event = frog_narrative_events.3
	}
	else_if = {
		limit = {
			global_var:frog_story_var ?= 3
		}
		trigger_event = frog_narrative_events.4
	}
	else_if = {
		limit = {
			global_var:frog_story_var ?= 4
		}
		trigger_event = frog_narrative_events.5
	}
	else = {
		error_log = "Effect frog_complete_investigation_effect was called, but frog_story_var did not have a valid value."
	}
}

frog_add_clues_effect = {
	custom_tooltip = {
		text = frog_add_clues_effect
		set_local_variable = {
			name = frog_add_clues_value
			value = $value$
		}
		frog_change_variable_effect = {
			NAME = frog_accumulated_clues_var
			AMOUNT = $value$
		}
	}
}

frog_add_progress_effect = {
	custom_tooltip = {
		text = frog_add_progress_effect
		set_local_variable = {
			name = frog_add_progress_value
			value = $value$
		}
		frog_change_variable_effect = {
			NAME = frog_investigation_progress_var
			AMOUNT = $value$
		}
	}
	frog_update_progress_bars_effect = yes
}

frog_add_peril_effect = {

		set_local_variable = {
			name = frog_add_peril_value
			value = $value$
		}
		frog_change_variable_effect = {
			NAME = frog_investigation_peril_var
			AMOUNT = $value$
		}
	}
	frog_update_progress_bars_effect = yes
}

frog_update_progress_bars_effect = {
	je:je_frog_noir_investigating_leads = {
		set_bar_progress = {
			name = frog_investigating_leads_progress_bar
			value = owner.var:frog_investigation_progress_var
		}
		set_bar_progress = {
			name = frog_investigating_leads_peril_bar
			value = owner.var:frog_investigation_peril_var
		}
	}
}

frog_save_sleuth_effect = {
	if = {
		limit = {
			frog_country_is_focused_on_finding_clues_trigger = yes
		}
		random_scope_character = {
			limit = {
				has_modifier = frog_character_is_official_investigator_modifier
			}
			save_scope_as = frog_investigator_scope
		}
	}
	else = {
		random_scope_character = {
			limit = {
				has_modifier = modifier_frog_sleuthing_for_clues
			}
			save_scope_as = frog_investigator_scope
		}
	}
}

frog_save_lead_investigator_effect = {
	random_scope_character = {
		limit = {
			has_modifier = frog_character_is_official_investigator_modifier
		}
		save_scope_as = frog_investigator_scope
	}
}

frog_save_watchman_effect = {
	random_scope_character = {
		limit = {
			has_modifier = modifier_frog_keeping_watch
		}
		save_scope_as = frog_watchman_scope
	}
}
