frog_sleuthing_for_clues_sgui = {
	scope = character
	is_shown = {
		has_modifier = modifier_frog_sleuthing_for_clues
	}
}

frog_start_sleuthing_for_clues_sgui = {
	scope = character
	is_valid = {
		NOT = {
			has_modifier = frog_character_is_official_investigator_modifier
		}
		frog_country_is_focused_on_finding_clues_trigger = no
	}
	effect = {
		if = {
			limit = {
				has_modifier = modifier_frog_sleuthing_for_clues
			}
			remove_modifier = modifier_frog_sleuthing_for_clues
		}
		else = {
			add_modifier = modifier_frog_sleuthing_for_clues
		}
		set_variable = {
			name = frog_action_var
			value = flag:modifier_frog_sleuthing_for_clues
		}
	}
}
