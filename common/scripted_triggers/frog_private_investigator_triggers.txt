frog_country_has_official_investigator_trigger = {
	custom_tooltip = {
		text = frog_country_has_official_investigator_trigger
		has_variable = frog_investigator_var
		exists = var:frog_investigator_var
	}
}

frog_country_is_focused_on_finding_clues_trigger = {
	custom_tooltip = {
		text = frog_country_is_focused_on_finding_clues_trigger
		var:frog_investigator_var ?= {
			has_modifier = modifier_frog_sleuthing_for_clues
		}
	}
}

frog_country_is_focused_on_following_leads_trigger = {
	custom_tooltip = {
		text = frog_country_is_focused_on_following_leads_trigger
		var:frog_investigator_var ?= {
			has_modifier = modifier_frog_following_leads
		}
	}
}

frog_character_has_personal_initative_trigger = {
	custom_tooltip = {
		text = frog_character_has_personal_initative_trigger
		NOT = {
			has_modifier = frog_character_is_official_investigator_modifier # The leader PI is tied to the firm
		}
		OR = {
			has_modifier = modifier_frog_sleuthing_for_clues
			has_modifier = modifier_frog_following_leads
			has_modifier = modifier_frog_keeping_watch
		}
	}
}

frog_any_character_is_keeping_watch_trigger = {
	custom_tooltip = {
		text = frog_any_character_is_keeping_watch_trigger
		any_scope_character = {
			has_modifier = modifier_frog_keeping_watch
		}
	}
}
