namespace = frog_frogged_by_toads_events

# Coastal Village Reports Toadmen Figures at Night
frog_frogged_by_toads_events.1 = {
	type = country_event
	placement = scope:frog_state_scope
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3
	dlc = dlc_frog_noir
	
	title = frog_frogged_by_toads_events.1.t
	desc = frog_frogged_by_toads_events.1.d
	flavor = frog_frogged_by_toads_events.1.f
	
	event_image = {
		video = "votp_gunboat_diplomacy"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		any_scope_state = {
			is_coastal = yes
		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_frog_noir_main
		}
	}
	
	cooldown = {
		years = 3
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				is_coastal = yes
			}
			save_scope_as = frog_state_scope
		}
		random_scope_character = {
			limit = {
				has_modifier = modifier_frog_keeping_watch
			}
			save_scope_as = frog_watchman_scope
		}
	}
	
	option = { # Leave the Villagers to be Spooked
		name = frog_frogged_by_toads_events.1.a
		default_option = yes
		scope:frog_state_scope = {
			add_modifier = {
				name = modifier_frog_midnight_croaking
				months = short_modifier_time
				is_decaying = yes
			}
		}
		frog_add_clues_effect = {
			value = 1
		}
	}
	
	option = { # Our Watchman is on the Case!
		name = frog_frogged_by_toads_events.1.b
		trigger = {
			exists = scope:frog_watchman_scope
		}
		frog_add_clues_effect = {
			value = 3
		}
	}
}

# Politician Approached by Toadmen
frog_frogged_by_toads_events.2 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_hardboiled.dds"
	duration = 3
	
	title = frog_frogged_by_toads_events.2.t
	desc = frog_frogged_by_toads_events.2.d
	flavor = frog_frogged_by_toads_events.2.f
	
	event_image = {
		video = "ep1_clandestine_meeting"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		any_scope_character = {
			has_role = character_role_politician
			prominence >= define:NCharacters|PROMINENCE_SIGNIFICANT_THRESHOLD
			NOR = {
				has_trait = pious
				has_trait = honorable
			}
		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_frog_noir_main
		}
	}
	
	cooldown = {
		years = 3
	}
	
	immediate = {
		random_scope_character = {
			limit = {
				has_role = character_role_politician
				prominence >= define:NCharacters|PROMINENCE_SIGNIFICANT_THRESHOLD
				NOR = {
					has_trait = pious
					has_trait = honorable
				}
			}
			save_scope_as = frog_character_scope
			
		}
		random_scope_character = {
			limit = {
				has_modifier = modifier_frog_keeping_watch
			}
			save_scope_as = frog_watchman_scope
		}
	}
	
	option = { # Mmm, I love corruption
		name = frog_frogged_by_toads_events.2.a
		default_option = yes
	}
	
	option = { # Break up the deal
		name = frog_frogged_by_toads_events.2.b
		trigger = {
			exists = scope:frog_watchman_scope
		}
		
	}
	
	option = { # Break the news to the press
		name = frog_frogged_by_toads_events.2.c
		trigger = {
			exists = scope:frog_watchman_scope
		}
	}
}

# Doppelganger at the Ball
frog_frogged_by_toads_events.3 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3
	
	title = frog_frogged_by_toads_events.3.t
	desc = frog_frogged_by_toads_events.3.d
	flavor = frog_frogged_by_toads_events.3.f
	
	event_image = {
		video = "europenorthamerica_art_gallery"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		any_scope_state = {
			state_has_building_type_levels = { # Can't have a high life if there's no where to have it
				target = bt:building_urban_center
				value >= 5
			}
		}
		any_scope_character = {
			is_interest_group_leader = yes
			prominence >= define:NCharacters|PROMINENCE_SIGNIFICANT_THRESHOLD
		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_frog_noir_main
		}
	}
	
	cooldown = {
		years = 3
	}
	
	immediate = {
		random_scope_character = {
			limit = {
				is_interest_group_leader = yes
				prominence >= define:NCharacters|PROMINENCE_SIGNIFICANT_THRESHOLD
			}
			save_scope_as = frog_character_scope
		}
	}
	
	option = { # Take notes
		name = frog_frogged_by_toads_events.3.a
		default_option = yes
		frog_add_clues_effect = {
			value = 1
		}
		scope:frog_character_scope = {
			add_modifier = modifier_frog_dating_a_toad
		}
	}
	
	option = { # Umm, sir, you're literally hitting on a toadgirl ( ͡° ͜ʖ ͡°)
		name = frog_frogged_by_toads_events.3.b
		trigger = {
			any_scope_character = {
				has_modifier = modifier_frog_keeping_watch
			}
		}
		show_as_unavailable = {
			always = yes
		}
		highlighted_option = yes
		frog_add_clues_effect = {
			value = 3
		}
		scope:frog_character_scope = {
			add_modifier = {
				name = modifier_frog_nearly_dated_a_toad
				months = long_modifier_time
			}
			interest_group = {
				add_modifier = {
					name = modifier_frog_exposed_toad_kink
					months = long_modifier_time
				}
			}
		}
	}
}

	event_image = {
		video = "southamerica_factory_opening"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		any_company = {
			prosperity >= 50
		}
	}

		NOT = {

			}
		}
	}
	
	option = { # Take Notes

		default_option = yes
	}
	
	option = { # Question the Executive
		name = frog_frogged_by_toads_events.4.b

	}
