# Slime Markings Spotted
frog_investigation_events.1 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	duration = 3
	
	title = frog_investigation_events.1.t
	desc = frog_investigation_events.1.d
	flavor = frog_investigation_events.1.f
	
	event_image = {
		video = "unspecific_gears_pistons"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		any_scope_state = {
			has_building = building_trade_center
		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_frog_noir_main
		}
	}
	
	cooldown = {
		years = 2
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				has_building = building_trade_center
			}
			save_scope_as = frog_state_scope
		}
		frog_save_sleuth_effect = yes
	}
	
	option = { # Perform a cursory search
		name = frog_investigation_events.1.a
		default_option = yes
		frog_add_clues_effect = {
			value = 2
		}
	}
	
	option = { # Shut down operations and take detailed noted
		name = frog_investigation_events.1.b
		trigger = {
			NOT = {
				has_law_or_variant = law_type:law_laissez_faire
			}
		}
		show_as_unavailable = {
			always = yes
		}
		frog_add_clues_effect = {
			value = 5
		}
		scope:frog_state_scope = {
			add_modifier = {
				name = modifier_frog_investigative_disruptions
				days = short_modifier_time
			}
		}
	}
}

		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_frog_noir_main
		}
	}
	
	cooldown = {
		years = 2
	}
	
	immediate = {
		random_scope_state = {
			limit = {
				has_building = building_urban_center
				state_unemployment_rate >= 0.02 # Gotta have some level of unemployment for there to be bums lying around
			}
			save_scope_as = frog_state_scope
		}
		frog_save_sleuth_effect = yes
	}
	
	option = { # Hmm, take notes
		name = frog_investigation_events.2.a
		default_option = yes
		frog_add_clues_effect = {
			value = 1
		}
	}
	
	option = { # Have the police "extract" more information
		name = frog_investigation_events.2.b
		trigger = {
			institution_investment_level = {
				institution = institution_police
				value >= 3
			}
		}
		show_as_unavailable = {
			always = yes
		}
		frog_add_clues_effect = {
			value = 3
		}
		scope:frog_state_scope = {
			add_radicals_in_state = {
				strata = lower
				value = small_radicals
			}
		}
	}
}

# Pointing Fingers
frog_investigation_events.5 = {
	type = country_event
	placement = root
	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"
	duration = 3
	
	title = frog_investigation_events.5.t
	desc = frog_investigation_events.5.d
	flavor = frog_investigation_events.5.f
	
	event_image = {
		video = "ep1_clandestine_meeting"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	
	trigger = {
		any_scope_character = {
			has_modifier = modifier_frog_sleuthing_for_clues
		}
		any_interest_group = {
			is_marginal = no
			ig_approval < 0
			NOT = {
				has_modifier = modifier_frog_finger_pointed_at
			}
		}
	}
	
	cancellation_trigger = {
		NOT = {
			has_global_variable = je_frog_noir_main
		}
	}
	
	cooldown = {
		years = 4
	}
	
	immediate = {
		random_interest_group = {
			limit = {
				is_marginal = no
				ig_approval < 0
				NOT = {
					has_modifier = modifier_frog_finger_pointed_at
				}
			}
			save_scope_as = frog_interest_group
		}
		frog_save_sleuth_effect = yes
	}
	
	option = {
		name = frog_investigation_events.5.a
		default_option = yes
		custom_tooltip = frog_nothing_happens
	}
	
	option = {
		name = frog_investigation_events.5.b
		scope:frog_interest_group = {
			add_modifier = {
				name = modifier_frog_finger_pointed_at
				days = normal_modifier_time
			}
		}
		add_radicals = {
			interest_group = scope:frog_interest_group
			value = medium_radicals
		}
		frog_add_clues_effect = {
			value = 3
		}
	}
}
